Supabase & MySQL Database Setup SQL

Run this SQL script in your database editor (Supabase SQL Editor or XAMPP phpMyAdmin):

-- ==========================================
-- OMNISTORE ENTERPRISE SUPABASE / MYSQL SCHEMA
-- ==========================================

-- 1. ACCOUNTS TABLE
CREATE TABLE IF NOT EXISTS accounts (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  password TEXT DEFAULT '123456',
  role TEXT CHECK (role IN ('Admin', 'Manager', 'Cashier')) DEFAULT 'Cashier',
  status TEXT DEFAULT 'Active',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. INVENTORY TABLE
CREATE TABLE IF NOT EXISTS inventory (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  sku TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  category TEXT,
  price NUMERIC(10,2) NOT NULL DEFAULT 0.00,
  cost_price NUMERIC(10,2) DEFAULT 0.00,
  stock INT DEFAULT 0,
  min_stock_alert INT DEFAULT 5,
  expiry_date TEXT DEFAULT 'N/A',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. SUPPLIERS TABLE
CREATE TABLE IF NOT EXISTS suppliers (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  company_name TEXT NOT NULL,
  contact_person TEXT,
  phone TEXT,
  category TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. TRANSACTIONS TABLE
CREATE TABLE IF NOT EXISTS transactions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  ref_number TEXT NOT NULL,
  payment_method TEXT DEFAULT 'Cash',
  items_summary TEXT,
  total_amount NUMERIC(10,2) NOT NULL DEFAULT 0.00,
  items_count INT NOT NULL DEFAULT 1,
  processed_by TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. TRANSACTION ITEMS TABLE
CREATE TABLE IF NOT EXISTS transaction_items (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  transaction_id UUID REFERENCES transactions(id) ON DELETE CASCADE,
  inventory_id UUID REFERENCES inventory(id),
  product_name TEXT NOT NULL,
  quantity INT DEFAULT 1,
  unit_price NUMERIC(10,2) DEFAULT 0.00,
  subtotal NUMERIC(10,2) DEFAULT 0.00,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. PAYROLL TABLE
CREATE TABLE IF NOT EXISTS payroll (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  employee_id UUID REFERENCES accounts(id) ON DELETE SET NULL,
  employee_name TEXT NOT NULL,
  hourly_rate NUMERIC(10,2) DEFAULT 0.00,
  worked_hours NUMERIC(10,2) DEFAULT 0.00,
  gross_salary NUMERIC(10,2) DEFAULT 0.00,
  deductions NUMERIC(10,2) DEFAULT 0.00,
  net_salary NUMERIC(10,2) DEFAULT 0.00,
  payroll_period TEXT,
  status TEXT DEFAULT 'Paid',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. SYSTEM AUDIT LOGS TABLE
CREATE TABLE IF NOT EXISTS system_logs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  time_stamp TIMESTAMPTZ DEFAULT NOW(),
  user_name TEXT NOT NULL,
  module TEXT NOT NULL,
  action_details TEXT NOT NULL
);

-- DISABLE ROW LEVEL SECURITY (RLS) FOR UNRESTRICTED PUBLIC DEVELOPMENT API ACCESS
ALTER TABLE accounts DISABLE ROW LEVEL SECURITY;
ALTER TABLE inventory DISABLE ROW LEVEL SECURITY;
ALTER TABLE suppliers DISABLE ROW LEVEL SECURITY;
ALTER TABLE transactions DISABLE ROW LEVEL SECURITY;
ALTER TABLE transaction_items DISABLE ROW LEVEL SECURITY;
ALTER TABLE payroll DISABLE ROW LEVEL SECURITY;
ALTER TABLE system_logs DISABLE ROW LEVEL SECURITY;

-- GRANT PUBLIC ANON AND AUTHENTICATED API PERMISSIONS
GRANT ALL ON ALL TABLES IN SCHEMA public TO anon, authenticated, service_role;
